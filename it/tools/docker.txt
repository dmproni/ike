#docker

----- how to make docker be accessible via http:
sudo bash -c 'mkdir -p /etc/docker && cat > /etc/docker/daemon.json <<EOF
{ "hosts": ["fd://", "tcp://0.0.0.0:2375"], "tls": false, "icc": false }
EOF'
sudo systemctl edit docker
[Service]
ExecStart=
ExecStart=/usr/bin/dockerd
sudo systemctl restart docker

----- how to install docker on debian 11
sudo apt update && sudo apt install -y ca-certificates curl gnupg lsb-release
sudo mkdir -p /etc/apt/keyrings && curl -fsSL https://download.docker.com/linux/debian/gpg | sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg
echo "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg] https://download.docker.com/linux/debian $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null
sudo apt update && sudo apt install -y docker-ce docker-ce-cli containerd.io
sudo systemctl start docker && sudo systemctl enable docker
sudo usermod -aG docker $USER && newgrp docker
